Google Apps Script Backend Code

Copy this code into your Google Sheet's Apps Script editor (`Extensions > Apps Script`) to power API calls.

// Code.gs - Backend API for HR Roster & Attendance Database

function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) || "getData";
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action === "getData") {
    return outputJSON({
      status: "success",
      employees: getSheetData(ss.getSheetByName("Employees")),
      shifts: getSheetData(ss.getSheetByName("Shifts Master")),
      roster: getSheetData(ss.getSheetByName("Roster Schedule")),
      attendance: getSheetData(ss.getSheetByName("Attendance Logs"))
    });
  }
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(data.sheetName);
    
    if (!sheet) {
      return outputJSON({ status: "error", message: "Sheet not found: " + data.sheetName });
    }
    
    sheet.appendRow(data.rowValues);
    return outputJSON({ status: "success", message: "Row added successfully" });
  } catch (err) {
    return outputJSON({ status: "error", message: err.toString() });
  }
}

function getSheetData(sheet) {
  if (!sheet) return [];
  var rows = sheet.getDataRange().getValues();
  var headers = rows[0];
  var data = [];
  
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = row[j];
    }
    data.push(obj);
  }
  return data;
}

function outputJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
